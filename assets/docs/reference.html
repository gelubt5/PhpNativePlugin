Example Usage

Using Colors 
use Colors;

// In simple.php functions
button("Save", "save", ['color' => Colors::SUCCESS]);
button("Delete", "delete", ['color' => Colors::DANGER]);
label("Warning!", ['color' => Colors::WARNING]);

// In ui_core.php fluent API
(new Button("Submit"))
    ->setBackground(Colors::PRIMARY)
    ->setTextColor(Colors::TEXT);

(new CardView())
    ->setBackground(Colors::CARD);
